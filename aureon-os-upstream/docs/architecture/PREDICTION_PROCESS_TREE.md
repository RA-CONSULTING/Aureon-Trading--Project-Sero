## 🔄 COMPLETE DECISION FLOW (Pseudocode)

```python
def should_execute_trade(symbol):
    """
    Complete decision flow using ALL prediction systems.
    """
    
    # ══════════════════════════════════════════════════════════════
    # LAYER 1: COSMIC GATE CHECKS
    # ══════════════════════════════════════════════════════════════
    
    # 1A. Earth Resonance Gate
    earth_state = earth_engine.compute_state()
    if earth_state.field_coherence < 0.50:
        return False, "Earth gate CLOSED - Low coherence"
    if earth_state.phase_lock_ratio < 0.60:
        return False, "Earth gate CLOSED - Phase lock failed"
    
    phi_multiplier = earth_state.gaia_pulse_phi  # 1.0-1.618
    
    # 1B. Cosmic State Check
    cosmic_state = cosmic_engine.compute_state()
    if cosmic_state.phase == CosmicPhase.DISTORTION:
        return False, "Cosmic gate CLOSED - Distortion phase"
    
    cosmic_boost = {
        'UNITY': 1.5,
        'COHERENCE': 1.3,
        'HARMONIC': 1.1,
        'TRANSITION': 0.8,
    }.get(cosmic_state.phase.name, 1.0)
    
    # 1C. Planetary Torque
    planetary_torque = cosmic_engine.compute_planetary_torque(datetime.now())
    
    # ══════════════════════════════════════════════════════════════
    # LAYER 2: PROBABILITY FORECAST (60 seconds ahead)
    # ══════════════════════════════════════════════════════════════
    
    # 2A. Collect price data (30 seconds)
    snapshots = collect_price_data(symbol, duration=30)
    
    # 2B. Generate probability matrix
    matrix = probability_analyzer.generate_matrix(symbol, current_data={
        'price': snapshots[-1].price,
        'momentum': compute_momentum(snapshots),
        'frequency': price_to_frequency(snapshots),
        'coherence': cosmic_state.coherence,
    })
    
    # 2C. Get fine-tuned forecast
    forecast = matrix.fine_tuned_probability  # Uses Hour +2 to refine Hour +1
    confidence = matrix.confidence_score
    
    if forecast < 0.65:
        return False, f"Probability too low: {forecast:.1%}"
    if confidence < 0.50:
        return False, f"Confidence too low: {confidence:.1%}"
    
    # ══════════════════════════════════════════════════════════════
    # LAYER 3: PROFIT CALCULATION
    # ══════════════════════════════════════════════════════════════
    
    current_price = snapshots[-1].price
    projected_change = compute_60s_projection(snapshots)
    
    fees = 0.001 * 2  # 0.1% × 2 (buy + sell) = 0.2%
    min_profit = fees + 0.0005  # + 0.05% minimum
    
    if projected_change < min_profit:
        return False, f"Projected change {projected_change:.3%} < minimum {min_profit:.3%}"
    
    # ══════════════════════════════════════════════════════════════
    # LAYER 4: POSITION SIZING
    # ══════════════════════════════════════════════════════════════
    
    base_position = 10.0  # $10 base
    
    position_size = (
        base_position
        × phi_multiplier        # Earth resonance (1.0-1.618)
        × cosmic_boost          # Cosmic phase (0.8-1.5)
        × planetary_torque      # Calendar torque (0.5-3.0)
        × matrix.position_modifier  # Probability (0.1-1.2)
    )
    
    # ══════════════════════════════════════════════════════════════
    # EXECUTE TRADE
    # ══════════════════════════════════════════════════════════════
    
    return True, {
        'action': 'BUY',
        'symbol': symbol,
        'position': position_size,
        'forecast_probability': forecast,
        'confidence': confidence,
        'projected_profit': projected_change - fees,
        'hold_time': 60,  # seconds
        'phi_boost': phi_multiplier,
        'cosmic_boost': cosmic_boost,
        'torque': planetary_torque,
    }
```
